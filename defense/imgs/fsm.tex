% Defence copy of thesis/imgs/fsm.tex, differing in ONE line: the on-site
% field arrow is drawn in the NN blue instead of teal, so the whole
% transverse-field Ising part of the machine carries a single colour and
% matches the colour-coded W_H matrix on slide 15. The thesis figure is
% untouched; \graphicspath puts defense/imgs/ first, so this one wins.
% Finite-state-machine diagram for the ANNNI-type Hamiltonian MPO, Equation (eq:Wannni).
% Each arrow a -> b is the tensor entry (W_H)_{ab}; every path 1 -> 5 is one Hamiltonian term.
%
% Standalone: compiles on its own to imgs/fsm.pdf.
% Inline:     with \usepackage{standalone} in the preamble, \input{imgs/fsm} drops it straight in.
\documentclass[tikz,border=3pt]{standalone}
\usepackage{amsmath,amssymb,bbold}
\usetikzlibrary{automata,arrows.meta,positioning}

% bbold's \mathbb{1}. Swap to \mathbbm{1} (package bbm) to match Equation (eq:Wannni) glyph for glyph.
\newcommand{\ident}{\mathbb{1}}

\begin{document}
\begin{tikzpicture}[
    >={Stealth[round,length=2.6mm]},
    shorten >=1pt,
    state/.style={circle, draw, thick, minimum size=9mm, inner sep=1pt},
    lbl/.style={font=\small, inner sep=2pt},
    ident/.style={draw=black!55, ->, thick},
    onsite/.style={draw=blue!65!black, ->, thick},
    nnz/.style={draw=blue!65!black, ->, thick},
    nnx/.style={draw=orange!80!black, ->, thick},
    nnn/.style={draw=violet!85!black, ->, thick},
  ]

  % Virtual states. 1 = nothing started, 2 = sigma^z endpoint pending,
  % 3 = sigma^x endpoint pending, 4 = NNN term waiting out the intermediate site, 5 = term closed.
  \node[state, initial below, initial text={start}] (s1) at (0,0) {$1$};
  \node[state] (s4) at (1.8,1.8) {$4$};
  \node[state] (s3) at (2.2,4.4) {$3$};
  \node[state] (s2) at (5.0,1.4) {$2$};
  \node[state, accepting] (s5) at (9,0) {$5$};

  % Identity loops: before the first term starts and after the last one closes.
  \path (s1) edge[ident, loop left]  node[lbl, left]  {$\ident$} (s1);
  \path (s5) edge[ident, loop right] node[lbl, right] {$\ident$} (s5);

  % On-site field, closed at the same site.
  \path (s1) edge[onsite] node[lbl, below] {$-\lambda\,\sigma^x$} (s5);

  % Nearest-neighbour sigma^z sigma^z, via the pending-z state.
  \path (s1) edge[nnz] node[lbl, below right, pos=0.55] {$-\sigma^z$} (s2);
  \path (s2) edge[nnz] node[lbl, above right] {$\sigma^z$} (s5);

  % Nearest-neighbour sigma^x sigma^x, via the pending-x state.
  \path (s1) edge[nnx] node[lbl, above left, pos=0.6] {$-p\lambda\,\sigma^x$} (s3);
  \path (s3) edge[nnx, bend left=12] node[lbl, above right] {$\sigma^x$} (s5);

  % Next-nearest neighbour: state 4 holds the term across the intermediate site, then rejoins 2.
  \path (s1) edge[nnn] node[lbl, below right, pos=0.60] {$-p\,\sigma^z$} (s4);
  \path (s4) edge[nnn] node[lbl, above] {$\ident$} (s2);

\end{tikzpicture}
\end{document}
